-- Run once in Supabase SQL Editor (after schema.sql) -- Public bucket for product photos; only admins can upload/delete insert into storage.buckets(id,name,public) values('product-images','product-images',true) on conflict do nothing; create policy "img read" on storage.objects for select using(bucket_id='product-images'); create policy "img admin insert" on storage.objects for insert with check(bucket_id='product-images' and public.is_admin()); create policy "img admin delete" on storage.objects for delete using(bucket_id='product-images' and public.is_admin()); -- Admin order status change (restores stock when cancelling) create or replace function admin_set_status(p_id uuid, p_status text, p_tracking text default null) returns void language plpgsql security definer set search_path=public as $$ declare o orders%rowtype; l jsonb; begin if not is_admin() then raise exception 'Not allowed'; end if; select * into o from orders where id=p_id for update; if not found then raise exception 'Order not found'; end if; if o.order_status='cancelled' and p_status<>'cancelled' then raise exception 'Cancelled orders cannot be reopened'; end if; if p_status='cancelled' and o.order_status<>'cancelled' then for l in select * from jsonb_array_elements(o.items) loop update products set stock=stock+(l->>'quantity')::int where id=(l->>'id')::uuid; end loop; end if; update orders set order_status=p_status, tracking_id=coalesce(nullif(p_tracking,''),tracking_id) where id=p_id; end $$; grant execute on function admin_set_status(uuid,text,text) to authenticated;
